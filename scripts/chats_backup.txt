import { useRouter } from 'expo-router';
import React from 'react';
import { View, Text, ScrollView, ActivityIndicator, Pressable, TextInput } from 'react-native';
import { Search, User, Filter, ArrowLeft } from 'lucide-react-native';
import { useChats } from '../src/api/client';

export default function ChatsScreen() {
  const router = useRouter();
  const { data: chats, isLoading } = useChats();

  return (
    <View className="flex-1 bg-[#F7F8FA] dark:bg-[#090A0C] pt-16">
      
      <View className="px-6 mb-4 flex-row items-center">
        <Pressable onPress={() => router.back()} className="mr-3 active:opacity-60">
          <ArrowLeft size={24} className="text-[#0F172A] dark:text-white" />
        </Pressable>
  
        <Text className="text-[#0F172A] dark:text-white text-3xl font-bold tracking-tight">Chats</Text>
        <Pressable className="w-10 h-10 rounded-full bg-white dark:bg-[#15171A] border border-gray-200 dark:border-[#1F2227] items-center justify-center shadow-sm dark:shadow-none">
          <Filter size={20} className="text-[#0F172A] dark:text-white" />
        </Pressable>
      </View>

      <View className="px-6 mb-6">
        <View className="flex-row items-center bg-white dark:bg-[#15171A] border border-gray-200 dark:border-[#1F2227] rounded-full px-4 py-2 h-12 shadow-[0_2px_10px_rgb(0,0,0,0.02)] dark:shadow-none">
          <Search size={20} className="text-[#64748B] dark:text-[#A1A1AA]" />
          <TextInput 
            placeholder="Search conversations..." 
            placeholderTextColor="#94A3B8"
            className="flex-1 ml-3 text-[#0F172A] dark:text-white text-base"
          />
        </View>
      </View>

      <ScrollView className="flex-1 px-4">
        {isLoading && (
          <View className="py-8 items-center justify-center">
            <ActivityIndicator size="large" color="#E2FF6E" />
          </View>
        )}

        {chats?.length === 0 && !isLoading && (
          <View className="py-8 items-center justify-center">
            <Text className="text-[#64748B] dark:text-[#A1A1AA]">No active chats found.</Text>
          </View>
        )}

        {chats?.map((chat: any) => (
          <Pressable 
            key={chat.id} 
            onPress={() => router.push(`/chat/${chat.id}`)}
            className={`bg-white dark:bg-[#15171A] p-4 rounded-3xl dark:rounded-[24px] mb-3 border ${chat.unread ? 'border-blue-500 dark:border-[#E2FF6E]' : 'border-gray-100 dark:border-[#1F2227]'} shadow-[0_4px_20px_rgb(0,0,0,0.03)] dark:shadow-none flex-row items-center active:opacity-80`}
          >
            <View className="w-12 h-12 rounded-full bg-blue-50 dark:bg-blue-900/20 items-center justify-center mr-4">
              <User size={24} className="text-blue-500 dark:text-blue-400" />
            </View>
            <View className="flex-1">
              <View className="flex-row justify-between items-center mb-1">
                <Text className="text-[#0F172A] dark:text-white font-bold text-base" numberOfLines={1}>
                  {chat.name}
                </Text>
                <Text className="text-[#64748B] dark:text-[#71717A] text-xs font-medium">
                  {chat.time}
                </Text>
              </View>
              <View className="flex-row items-center">
                <Text className={`${chat.unread ? 'text-[#0F172A] dark:text-white font-semibold' : 'text-[#64748B] dark:text-[#A1A1AA]'} text-sm flex-1 mr-2`} numberOfLines={1}>
                  {chat.message}
                </Text>
                {chat.unread > 0 && (
                  <View className="w-5 h-5 rounded-full bg-blue-500 dark:bg-[#E2FF6E] items-center justify-center">
                    <Text className="text-white dark:text-[#090A0C] text-[10px] font-bold">{chat.unread}</Text>
                  </View>
                )}
              </View>
            </View>
          </Pressable>
        ))}
      </ScrollView>
    </View>
  );
}
